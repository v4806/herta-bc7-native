cmake_minimum_required(VERSION 3.20)
project(herta_bc7f_pipeline LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This experimental C ABI currently targets Windows x64 with MSVC")
endif()
set(BASIS_UNIVERSAL_ROOT "" CACHE PATH "Basis Universal checkout at ee771a7f913a3233a252eb6d281b4446f6a3c9f5")
if(NOT EXISTS "${BASIS_UNIVERSAL_ROOT}/transcoder/basisu_transcoder.cpp")
  message(FATAL_ERROR "Set BASIS_UNIVERSAL_ROOT to the pinned Basis Universal checkout")
endif()
add_library(herta_pipeline_windows_x64 SHARED pipeline.cpp bc7f_bridge.cpp
  "${BASIS_UNIVERSAL_ROOT}/transcoder/basisu_transcoder.cpp")
target_include_directories(herta_pipeline_windows_x64 PRIVATE "${BASIS_UNIVERSAL_ROOT}/transcoder")
target_compile_features(herta_pipeline_windows_x64 PRIVATE cxx_std_17)
target_compile_definitions(herta_pipeline_windows_x64 PRIVATE BASISD_SUPPORT_KTX2_ZSTD=0 _CRT_SECURE_NO_WARNINGS)
target_compile_options(herta_pipeline_windows_x64 PRIVATE /O2 /utf-8 /EHsc)
set_property(TARGET herta_pipeline_windows_x64 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
option(HERTA_AVX2 "Build for AVX2 CPUs, matching the measured local binary" ON)
if(HERTA_AVX2)
  target_compile_options(herta_pipeline_windows_x64 PRIVATE /arch:AVX2)
endif()
